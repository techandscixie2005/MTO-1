    if v.size == 0:
        x_dense = np.linspace(bins[0], bins[-1], dense_pts)
        return x_dense, np.zeros_like(x_dense)
    centers = (bins[:-1] + bins[1:]) / 2.0
    hist, _ = np.histogram(v, bins=bins)
    total = hist.sum()

    y_smooth = gaussian_filter1d(y_dense, sigma=sigma)
    return x_dense, np.clip(y_smooth, 0.0, None)

    hist, _ = np.histogram(v, bins=bins)
    total = hist.sum()

    y_smooth = gaussian_filter1d(y_dense, sigma=sigma)

def load_smiles(txt_file):
    with open(txt_file, "r") as f:
        smiles_list = [line.strip() for line in f.readlines()]
    return smiles_list

# **3. 转换 SMILES 为凯库勒式**
def smiles_to_kekule(smiles):
    mol = Chem.MolFromSmiles(smiles)
    if mol is not None:
        kekule_smiles = Chem.MolToSmiles(mol, kekuleSmiles=True)
        return kekule_smiles
    return None

# **4. 解析 SMILES 并构建 SMILES_VOCAB**
def tokenize_smiles(smiles_list):
    # SMILES 正则表达式（确保 Cl 和 Br 作为单独 token）
    pattern = r"(\[[^\]]+]|Br?|Cl?|N|O|S|P|F|I|b|c|n|o|s|p|\(|\)|\.|=|#|-|\+|\\|\/|:|~|@|\?|>|\*|\$|\%[0-9]{2}|[0-9])"

    tokenized_smiles = []
    vocab_set = set()

    for smiles in smiles_list:
        # 转换为凯库勒式
        kekule_smiles = smiles_to_kekule(smiles)
        if kekule_smiles is None:
            continue  # 忽略无法解析的 SMILES

        # 使用正则表达式进行拆分
        tokens = re.findall(pattern, kekule_smiles)
        tokenized_smiles.append(tokens)

        # 记录所有独特的 token
        vocab_set.update(tokens)

    return tokenized_smiles, vocab_set

# **9. Normalize each spectrum individually and set values below 1e-5 to 1e-5**
def normalize_spectra_individual(spectra):
    # Normalize each spectrum independently
    normalized_spectra = []
    for spectrum in spectra:
        min_val = np.min(spectrum)
        max_val = np.max(spectrum)
        
        # Ensure min_val != max_val to avoid division by zero
        if max_val != min_val:
            normalized_spectrum = (spectrum - min_val) / (max_val - min_val)
        else:
            normalized_spectrum = spectrum  # In case all values are the same

        # Set small values to 1e-5
        normalized_spectrum[normalized_spectrum < 1e-5] = 1e-5
        normalized_spectra.append(normalized_spectrum)

    return np.array(normalized_spectra)

def hist_to_percent_curve(values, bins, sigma=2.0, dense_pts=1200):
    v = np.asarray(values)
    v = v[~np.isnan(v)]
    if v.size == 0:
        x_dense = np.linspace(bins[0], bins[-1], dense_pts)
        return x_dense, np.zeros_like(x_dense)
    centers = (bins[:-1] + bins[1:]) / 2.0
    hist, _ = np.histogram(v, bins=bins)
    total = hist.sum()
    pct = (hist / total * 100.0) if total > 0 else np.zeros_like(hist, dtype=float)
    x_dense = np.linspace(bins[0], bins[-1], dense_pts)
    y_dense = interp1d(centers, pct, kind='linear', bounds_error=False, fill_value=0.0)(x_dense)
    y_smooth = gaussian_filter1d(y_dense, sigma=sigma)
    return x_dense, np.clip(y_smooth, 0.0, None)

    """加载CSV文件并跳过指定行列"""

        if usecols:
            df = pd.read_csv(file_path, skiprows=skiprows, usecols=usecols)

def preprocess_data(data, method='quantile'):
    """数据预处理函数"""
    try:
        if method == 'quantile':
            # 分位数归一化（保留0值）
            transformer = QuantileTransformer(output_distribution='normal', random_state=42)
            processed = transformer.fit_transform(data)
            # 保持原始0值位置
            processed[data == 0] = 0
            
        elif method == 'log':
            # 对数变换（加极小值处理0值）
            processed = np.log1p(data)
            
        elif method == 'robust':
            # RobustScaler缩放
            scaler = RobustScaler()
            processed = scaler.fit_transform(data)
            
        elif method == 'power':
            # Box-Cox变换（需要正值）
            transformer = PowerTransformer(method='box-cox')
            processed = transformer.fit_transform(data)
            
        else:
            raise ValueError("无效的预处理方法")
            
        return processed
    
    except Exception as e:
        print(f"数据预处理失败：{str(e)}")
        exit(1)

    """保存处理后的数据"""
    try:
        pd.DataFrame(data).to_csv(output_file, index=False)

        print(f"保存数据时发生错误：{str(e)}")

def hist_to_percent_curve(values, bins, sigma=2.0, dense_pts=1200):
    v = np.asarray(values)
    v = v[~np.isnan(v)]
    if v.size == 0:
        x_dense = np.linspace(bins[0], bins[-1], dense_pts)
        return x_dense, np.zeros_like(x_dense)
    centers = (bins[:-1] + bins[1:]) / 2.0
    hist, _ = np.histogram(v, bins=bins)
    total = hist.sum()
    pct = (hist / total * 100.0) if total > 0 else np.zeros_like(hist, dtype=float)
    x_dense = np.linspace(bins[0], bins[-1], dense_pts)
    y_dense = interp1d(centers, pct, kind='linear', bounds_error=False, fill_value=0.0)(x_dense)
    y_smooth = gaussian_filter1d(y_dense, sigma=sigma)
    return x_dense, np.clip(y_smooth, 0.0, None)

def process_single_sample(data, model, device):
    """
    处理单个样本，提取原子数并计算推断时间
    
    Args:
        data: 单个样本数据
        model: 训练好的模型
        device: 计算设备
    
    Returns:
        dict: 包含原子数和推断时间的结果
    """
    result = {
        '原子数': -1,
        '推断时间_秒': 0.0,
        '推断时间_毫秒': 0.0,
        '状态': '失败'
    }
    
    try:
        # 1. 提取原子数（从z属性）
        if hasattr(data, 'z'):
            # z是原子序数数组，长度就是原子数
            num_atoms = data.z.shape[0]
            result['原子数'] = num_atoms
            
            # 可选：记录原子类型分布
            atom_types = data.z.tolist()
            unique_atoms = set(atom_types)
            result['原子类型'] = str(sorted(list(unique_atoms)))
        else:
            result['状态'] = '缺少z属性'
            return result
        
        # 2. 准备输入数据
        z = data.z.to(device)
        pos = data.pos.to(device)
        batch = torch.zeros(z.shape[0], dtype=torch.long, device=device)
        
        # 3. 计时推断
        if torch.cuda.is_available():
            torch.cuda.synchronize()
        
        start_time = time.perf_counter()
        
        with torch.no_grad():
            pred = model(z=z, pos=pos, batch=batch)
            # 确保预测值非负
            pred = torch.clamp(pred, min=0.0)
        
        if torch.cuda.is_available():
            torch.cuda.synchronize()
        
        infer_time = time.perf_counter() - start_time
        
        # 4. 记录结果
        result['推断时间_秒'] = infer_time
        result['推断时间_毫秒'] = infer_time * 1000
        result['状态'] = '成功'
        
        # 可选：记录预测光谱的统计信息
        pred_np = pred.detach().cpu().numpy().flatten()
        result['光谱长度'] = len(pred_np)
        result['光谱均值'] = float(np.mean(pred_np))
        result['光谱最大值'] = float(np.max(pred_np))
        
    except Exception as e:
        result['状态'] = f'错误: {str(e)[:50]}'
        # 如果开始计时但出错了，仍然记录时间
        if 'start_time' in locals():
            if torch.cuda.is_available():
                torch.cuda.synchronize()
            infer_time = time.perf_counter() - start_time
            result['推断时间_秒'] = infer_time
            result['推断时间_毫秒'] = infer_time * 1000
    
    return result

def hist_to_percent_curve(values, bins, sigma=2.0, dense_pts=1200):
    v = np.asarray(values)
    v = v[~np.isnan(v)]
    if v.size == 0:
        x_dense = np.linspace(bins[0], bins[-1], dense_pts)
        return x_dense, np.zeros_like(x_dense)
    centers = (bins[:-1] + bins[1:]) / 2.0
    hist, _ = np.histogram(v, bins=bins)
    total = hist.sum()
    pct = (hist / total * 100.0) if total > 0 else np.zeros_like(hist, dtype=float)
    x_dense = np.linspace(bins[0], bins[-1], dense_pts)
    y_dense = interp1d(centers, pct, kind='linear', bounds_error=False, fill_value=0.0)(x_dense)
    y_smooth = gaussian_filter1d(y_dense, sigma=sigma)
    return x_dense, np.clip(y_smooth, 0.0, None)

# ========== 从 shielding 和 data 推导化学位移 ==========
def compute_shifts_from_data(shielding, data, ref_c, ref_h):
    shielding = shielding.detach().cpu().numpy().astype(float).reshape(-1)
    Z = infer_Z_from_data(data)

# ========== 高斯展宽生成谱图 ==========
def gaussian_spectrum(x, shifts, sigma):
    y = np.zeros_like(x)
    for s in shifts:
        y += norm.pdf(x, loc=s, scale=sigma)
    if y.max() > 0:
        y = y / y.max()  # 归一化
    return y

def hist_to_percent_curve(values, bins, sigma=2.0, dense_pts=1200):
    v = np.asarray(values)
    v = v[~np.isnan(v)]
    if v.size == 0:
        x_dense = np.linspace(bins[0], bins[-1], dense_pts)
        return x_dense, np.zeros_like(x_dense)
    centers = (bins[:-1] + bins[1:]) / 2.0
    hist, _ = np.histogram(v, bins=bins)
    total = hist.sum()
    pct = (hist / total * 100.0) if total > 0 else np.zeros_like(hist, dtype=float)
    x_dense = np.linspace(bins[0], bins[-1], dense_pts)
    y_dense = interp1d(centers, pct, kind='linear', bounds_error=False, fill_value=0.0)(x_dense)
    y_smooth = gaussian_filter1d(y_dense, sigma=sigma)
    return x_dense, np.clip(y_smooth, 0.0, None)
